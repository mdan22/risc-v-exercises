# n = 5
# trinum(trinum(n))


# Assembler

main:
  # initialize n
  addi a0, x0, 5

  # store ra on stack
  addi sp, sp, -4
  sw ra, 0(sp)

  jal ra, trinum
  jal ra, trinum

  # restore ra from stack
  lw ra, 0(sp)
  addi sp, sp, 4
  
  jal x0, end

trinum:
  # store temp on stack
  addi sp, sp, -4
  sw t0, 0(sp)

  # initialize temp
  addi t0, a0, 0

  # actual function
  loop:
  addi t0, t0, -1
  beq t0, x0, loop_end
  add a0, a0, t0
  jal x0, loop
  loop_end:

  # restore temp from stack
  lw t0, 0(sp)
  addi sp, sp, 4

  # jump back
  jalr x0, ra, 0

end:
# whatever comes next
addi x0, x0, 0


Aufgabe 1c)

# Pseudocode

# trinum(int n) {
#   if (n == 1) {
#     return 1;
#   }
#   else {
#     return n + trinum(n-1);
#   }
# }


# Assembler

main:

  # store temporaries on stack
  addi sp, sp, -8
  sw t0, 4(sp)
  sw t1, 0(sp)

  # initialize n
  addi a0, x0, 4

  # trinum function call with n = 4
  jal ra, trinum

  # restore temporaries from stack
  lw t1, 0(sp)
  lw t0, 4(sp)
  addi sp, sp, 8

  jal x0, end

trinum:
  # store ra, a0 on stack
  addi sp, sp, -8
  sw ra, 4(sp)
  sw a0, 0(sp)

  addi t0, a0, -1 # store n-1 in t0 for bne
  bne t0, x0, L1 # base case: if (n = 1) -> termination condition at n = 1
  addi a0, x0, 1 # return 1
  addi sp, sp, 8 # pop stack
  jalr x0, ra, 0 # jmp back to ra of last recursive caller

  L1:
  addi a0, a0, -1
  jal ra, trinum
  addi t1, a0, 0

  # restore a0, ra from stack
  lw a0, 0(sp)
  lw ra, 4(sp)
  addi sp, sp, 8

  add a0, t1, a0
  jalr x0, ra, 0

end:
addi x0, x0, 0 # whatever comes next

sb t6, 0(t2)

loop:

  addi t2, t2, 1

    addi t1, x0, 69 # 'E' = 69
    jal x0, loop

  addi t1, x0, 0
